Google Sheets sync Not configured

📋 Sheets setup (~10 min)

  1. Go to sheets.google.com → create a sheet: Label Studio
  2. Click Extensions → Apps Script
  3. Delete existing code, paste the Sheets script below
  4. Save (💾) → Deploy → New deployment → Web app
  5. Execute as: Me · Who has access: Anyone
  6. Authorise with your Google account → copy the Web app URL
  7. Paste above → tap Test connection
const SHEET = 'Catalogue';
function doPost(e) {
  try {
    const d = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sh = ss.getSheetByName(SHEET);
    if (!sh) {
      sh = ss.insertSheet(SHEET);
      sh.appendRow(['SKU','Category','Name','Brand','Colour',
        'Condition','Size','Listed Price','Cost Price',
        'Notes','Platforms','Photo URL','Date Added']);
      sh.getRange(1,1,1,13).setFontWeight('bold');
    }
    sh.appendRow([d.sku,d.catLabel,d.name,d.brand||'',d.colour||'',
      d.condition||'',d.size||'',d.price||'',d.cost||'',d.notes||'',
      (d.platforms||[]).join(', '),d.driveUrl||'',d.dateAdded||'']);
    return ContentService
      .createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok:false,error:err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
function doGet() {
  return ContentService
    .createTextOutput(JSON.stringify({ok:true,status:'connected'}))
    .setMimeType(ContentService.MimeType.JSON);
}

Photos save to "Label Studio Photos" in Google Drive. File name = SKU (e.g. SHO-001.jpg).

📋 Drive setup (~5 min)

  1. Go to script.google.com → New project
  2. Paste the Drive script below → Save
  3. Deploy → New deployment → Web app
  4. Same settings: Execute as Me, access Anyone
  5. Copy URL → paste above → tap Test connection
const FOLDER = 'Label Studio Photos';
function doPost(e) {
  try {
    const d = JSON.parse(e.postData.contents);
    const base64 = d.imageData.split(',')[1];
    const blob = Utilities.newBlob(
      Utilities.base64Decode(base64),
      d.mimeType || 'image/jpeg', d.filename);
    let folder;
    const folders = DriveApp.getFoldersByName(FOLDER);
    folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(FOLDER);
    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    const url = 'https://drive.google.com/uc?id=' + file.getId();
    return ContentService
      .createTextOutput(JSON.stringify({ok:true,url:url}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok:false,error:err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
function doGet() {
  return ContentService
    .createTextOutput(JSON.stringify({ok:true,status:'drive-connected'}))
    .setMimeType(ContentService.MimeType.JSON);
}
